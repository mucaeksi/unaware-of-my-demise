var base: String = ProjectSettings.globalize_path("res://").trim_suffix("/").get_base_dir().path_join("narrative/15_Art_Workbench/enemies/design_v001")
var source: Dictionary = JSON.parse_string(FileAccess.get_file_as_string(base.path_join("design_cards.json"))) as Dictionary
var window: Window = load("res://scripts/art_review/preview_session.gd").open(get_tree()) as Window
window.title = "UOMD · Enemy Design Candidates · Native Review"
var previous: Node = window.get_node_or_null("EnemyDesignReview")
if previous != null:
	window.remove_child(previous)
	previous.queue_free()
for child: Node in window.get_children():
	if child is Control:
		if not child.has_meta("enemy_review_previous_visible"):
			child.set_meta("enemy_review_previous_visible", child.visible)
		child.hide()
var panel: Control = Control.new()
panel.name = "EnemyDesignReview"
window.add_child(panel)
panel.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
var bg: ColorRect = ColorRect.new()
bg.color = Color("151319")
panel.add_child(bg)
bg.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
var heading: Label = Label.new()
heading.text = "PART I · ENEMY DESIGN REVIEW"
heading.position = Vector2(24, 24)
heading.add_theme_font_size_override("font_size", 38)
panel.add_child(heading)
var subtitle: Label = Label.new()
subtitle.text = "26 candidates · 20 enemies + 6 bosses · Yseldra / Cael reserved"
subtitle.position = Vector2(24, 82)
subtitle.add_theme_font_size_override("font_size", 25)
panel.add_child(subtitle)
var tabs: TabContainer = TabContainer.new()
panel.add_child(tabs)
tabs.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
tabs.offset_left = 16.0
tabs.offset_top = 142.0
tabs.offset_right = -16.0
tabs.offset_bottom = -20.0
tabs.add_theme_font_size_override("font_size", 29)
var rows: Array[Dictionary] = []
var failures: Array[String] = []
DirAccess.make_dir_recursive_absolute(base.path_join("validation/native"))
for chapter: int in range(1, 8):
	var cards: Array[Dictionary] = []
	for value: Variant in source["cards"]:
		var card: Dictionary = value as Dictionary
		if int(card["first_design_chapter"]) == chapter:
			cards.append(card)
	var image: Image = Image.new()
	var path: String = base.path_join(str(cards[0]["board_path"]))
	var decode_error: int = image.load(path)
	if decode_error != OK:
		failures.append(path)
		continue
	var texture: ImageTexture = ImageTexture.create_from_image(image)
	var page: Control = Control.new()
	page.name = "Ch %d" % chapter
	tabs.add_child(page)
	var board: TextureRect = TextureRect.new()
	board.expand_mode = TextureRect.EXPAND_IGNORE_SIZE
	board.stretch_mode = TextureRect.STRETCH_KEEP_ASPECT_CENTERED
	board.texture = texture
	board.position = Vector2(16, 20)
	board.size = Vector2(1000, 580)
	page.add_child(board)
	var label: Label = Label.new()
	label.text = "STATIC CONCEPT VIEWS · NO ALPHA / ANIMATION CLAIM"
	label.position = Vector2(22, 622)
	label.add_theme_font_size_override("font_size", 26)
	page.add_child(label)
	for index: int in range(cards.size()):
		var atlas: AtlasTexture = AtlasTexture.new()
		atlas.atlas = texture
		var cell_width: float = float(image.get_width()) / float(cards.size())
		var upper_ratio: float = 0.615 if chapter == 6 else 0.60
		var region_x: float = float(index) * cell_width
		if chapter == 7:
			# Editorial board columns are unequal. These are review-only regions,
			# not production sprite-atlas cells or animation frames.
			var bounds: Array[float] = [0.0, 441.0, 887.0, 1257.0, 1774.0]
			region_x = bounds[index]
			cell_width = bounds[index + 1] - region_x
		atlas.region = Rect2(region_x, 0.0, cell_width, float(image.get_height()) * upper_ratio)
		atlas.filter_clip = true
		var view: TextureRect = TextureRect.new()
		view.expand_mode = TextureRect.EXPAND_IGNORE_SIZE
		view.stretch_mode = TextureRect.STRETCH_KEEP_ASPECT_CENTERED
		view.texture = atlas
		view.position = Vector2(20 + (index % 2) * 505, 686 + floori(float(index) / 2.0) * 456)
		view.size = Vector2(490, 408)
		page.add_child(view)
		var id_label: Label = Label.new()
		id_label.text = str(cards[index]["enemy_id"])
		id_label.position = view.position + Vector2(8, 410)
		id_label.add_theme_font_size_override("font_size", 23)
		page.add_child(id_label)
	tabs.current_tab = tabs.get_tab_count() - 1
	await get_tree().process_frame
	await RenderingServer.frame_post_draw
	var output: String = "validation/native/ch%02d_native.png" % chapter
	var save_error: int = window.get_texture().get_image().save_png(base.path_join(output))
	rows.append({"chapter": chapter, "enemy_count": cards.size(), "source": cards[0]["board_path"], "width": image.get_width(), "height": image.get_height(), "decode_error": decode_error, "save_error": save_error, "capture": output})
tabs.current_tab = 4
window.show()
return {"godot": Engine.get_version_info()["string"], "view": "reused native Local UI Preview; static board and temporary AtlasTexture concept crops only", "captures": rows, "failures": failures, "runtime_bindings_modified": false, "profile_modified": false, "alpha_sprite_acceptance": "PENDING", "battle_animation_acceptance": "PENDING"}
